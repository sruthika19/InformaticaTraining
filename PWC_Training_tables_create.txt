CREATE TABLE CUSTOMER
(
  CUSTOMER_ID NUMBER(10) PRIMARY KEY,
  CUSTOMER_NAME VARCHAR2(100) NOT NULL,
  DATE_OF_BIRTH DATE,
  CUSTOMER_TYPE VARCHAR2(20),
  RISK_CATEGORY VARCHAR2(10),
  EMAIL_ADDRESS VARCHAR2(100),
  MOBILE_NUMBER VARCHAR2(20),
  CREATED_DATE DATE,
  UPDATED_DATE DATE,
  CUSTOMER_STATUS VARCHAR2(10)
);

CREATE TABLE CUSTOMER_ADDRESS
(
  ADDRESS_ID NUMBER(10) PRIMARY KEY,
  CUSTOMER_ID NUMBER(10),
  ADDRESS_TYPE VARCHAR2(15),
  ADDRESS_LINE1 VARCHAR2(100),
  ADDRESS_LINE2 VARCHAR2(100),
  CITY VARCHAR2(50),
  STATE_NAME VARCHAR2(50),
  POSTAL_CODE VARCHAR2(15),
  COUNTRY_CODE VARCHAR2(3),
  VALID_FROM DATE,
  VALID_TO DATE
);

CREATE TABLE ACCOUNT
(
  ACCOUNT_ID NUMBER(10) PRIMARY KEY,
  CUSTOMER_ID NUMBER(10),
  BRANCH_ID NUMBER(6),
  ACCOUNT_TYPE VARCHAR2(20),
  CURRENCY_CODE VARCHAR2(3),
  OPEN_DATE DATE,
  CLOSE_DATE DATE,
  CURRENT_BALANCE NUMBER(18,2),
  CREDIT_LIMIT NUMBER(18,2),
  ACCOUNT_STATUS VARCHAR2(15),
  UPDATED_DATE DATE
);

CREATE TABLE ACCOUNT_TRANSACTION
(
  TRANSACTION_ID NUMBER(15) PRIMARY KEY,
  ACCOUNT_ID NUMBER(12),
  TRANSACTION_DATE DATE,
  TRANSACTION_TYPE VARCHAR2(10),
  TRANSACTION_AMOUNT NUMBER(18,2),
  CURRENCY_CODE VARCHAR2(3),
  CHANNEL VARCHAR2(20),
  DESCRIPTION VARCHAR2(200),
  REFERENCE_NUMBER VARCHAR2(50),
  TRANSACTION_STATUS VARCHAR2(15),
  CREATED_TIMESTAMP TIMESTAMP
);

CREATE TABLE LOAS
(
  LOAN_ID NUMBER(12) PRIMARY KEY,
  BRANCH_ID NUMBER(6),
  LOAN_TYPE VARCHAR2(30),
  ORIGINAL_AMOUNT NUMBER(18,2),
  OUTSTANDING_AMOUNT NUMBER(18,2),
  INTEREST_RATE NUMBER(8,4),
  START_DATE DATE
);
